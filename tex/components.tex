\makeatother

% Vector response marks avoid font-dependent checkbox glyphs.
\newcommand{\SingleMark}{\tikz[baseline=-.6ex]\draw[line width=.55pt] (0,0) circle[radius=1.5mm];}
\newcommand{\MultiMark}{\tikz[baseline=-.6ex]\draw[line width=.55pt] (0,0) rectangle (3mm,3mm);}
\newcommand{\CurrentMark}{\SingleMark}
\newsavebox{\QuestionBox}
\newcommand{\CurrentQuestionId}{}
\newenvironment{QuestionBlock}[4]{%
  \renewcommand{\CurrentQuestionId}{#1}%
  \begin{lrbox}{\QuestionBox}\begin{minipage}{\linewidth}
  \setlength{\parindent}{0pt}\setlength{\parskip}{0pt}%
  \ifSurveyRTL\raggedleft\else\RaggedRight\fi
  \label{q:#1:start}%
  {\bfseries #2.\enspace\tr{#3}\par}%
  \vspace{1pt}%
  {\fontsize{9}{11}\selectfont\color{black!70}\tr{#4}\par}%
  \vspace{3pt}%
}{%
  \label{q:\CurrentQuestionId:end}%
  \end{minipage}\end{lrbox}%
  \ifdim\dimexpr\ht\QuestionBox+\dp\QuestionBox\relax>\textheight
    \PackageError{survey}{Question \CurrentQuestionId\space is taller than a column}{Shorten this question or its answers.}%
  \fi
  \noindent\usebox{\QuestionBox}\par\vspace{10pt}%
}
